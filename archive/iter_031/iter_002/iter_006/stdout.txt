def calc_var_loss(x, gamma=1.0, eps=1e-4):
    mean = x.mean(dim=0)
    var = torch.mean((x - mean) ** 2, dim=0)
    std = torch.sqrt(var + eps)
    return torch.mean(F.relu(gamma - std))


def calc_cov_loss(x):
    B, d = x.shape
    if B <= 1 or d <= 1:
        return torch.tensor(0.0, device=x.device, dtype=x.dtype)
    mean = x.mean(dim=0, keepdim=True)
    x_centered = x - mean
    cov = (x_centered.T @ x_centered) / (B - 1)
    diag = torch.diagonal(cov)
    off_diag = cov - torch.diag(diag)
    return (off_diag ** 2).sum() / d

    Deconvolutional decoder head mapping (B, d_max, 8) -> (B, 3, 128).
    """

    def __init__(self, d_max=8):
        super().__init__()
        self.d_max = d_max
        self.deconv1 = nn.ConvTranspose1d(d_max, 128, kernel_size=5, stride=2, padding=2, output_padding=1)
        self.deconv2 = nn.ConvTranspose1d(128, 64, kernel_size=5, stride=2, padding=2, output_padding=1)
        self.deconv3 = nn.ConvTranspose1d(64, 32, kernel_size=5, stride=2, padding=2, output_padding=1)
        self.deconv4 = nn.ConvTranspose1d(32, 3, kernel_size=5, stride=2, padding=2, output_padding=1)

        """
        Args:
            a_dyn (Tensor): shape (B, d_max, 8)
        Returns:

class SeparateDynEncoder(nn.Module):
    """
    Encoder with TWO independent CNN backbones:
      (a) coord_backbone: conv1->conv2->conv3->conv4->conv_spatial
          Produces z_coord via soft-argmax centroid from spatial feature map.
      (b) dyn_backbone: conv1_dyn->conv2_dyn->conv3_dyn->conv4_dyn->conv_identity_dyn
          Same architecture, SEPARATE parameters.
          Produces z_dyn via mean pooling over the spatial dimension.

    Both backbones process the SAME RGB input independently.
    Each backbone adds positional encoding independently.
    """

    def __init__(self, d_max=8, pos_encoding="none", dyn_readout="mean",
                 sub_features=1, dyn_source="spatial"):
        super().__init__()
        self.d_max = d_max
        self.pos_encoding = pos_encoding
        # dyn_readout and sub_features accepted for interface compatibility,
        # but this encoder always uses mean readout on the dyn backbone.
        self.dyn_readout = dyn_readout
        self.sub_features = sub_features
        self.dyn_source = dyn_source

        if pos_encoding == "none":
            in_channels = 3
        elif pos_encoding == "linear":
            in_channels = 4
        elif pos_encoding == "sinusoidal":
            in_channels = 7
        else:
            raise ValueError(f"Unknown pos_encoding: {pos_encoding}")

        # --- Coord backbone ---
        self.conv1 = nn.Conv1d(in_channels, 16, kernel_size=5, stride=2, padding=2)
        self.conv2 = nn.Conv1d(16, 32, kernel_size=5, stride=2, padding=2)
        self.conv3 = nn.Conv1d(32, 64, kernel_size=5, stride=2, padding=2)
        self.conv4 = nn.Conv1d(64, 128, kernel_size=5, stride=2, padding=2)
        self.conv_spatial = nn.Conv1d(128, d_max, kernel_size=1)

        # --- Dyn backbone (independent parameters) ---
        self.conv1_dyn = nn.Conv1d(in_channels, 16, kernel_size=5, stride=2, padding=2)
        self.conv2_dyn = nn.Conv1d(16, 32, kernel_size=5, stride=2, padding=2)
        self.conv3_dyn = nn.Conv1d(32, 64, kernel_size=5, stride=2, padding=2)
        self.conv4_dyn = nn.Conv1d(64, 128, kernel_size=5, stride=2, padding=2)
        self.conv_identity_dyn = nn.Conv1d(128, d_max, kernel_size=1)

    @property
    def d_dyn(self):
        return self.d_max * self.sub_features

    def _forward_coord_backbone(self, x):
        """Returns backbone features (B, 128, 8) for coord stream."""
        x = add_positional_encoding(x, self.pos_encoding)
        x = F.relu(self.conv1(x))
        x = F.relu(self.conv2(x))
        x = F.relu(self.conv3(x))
        x = F.relu(self.conv4(x))
        return x

    def _forward_dyn_backbone(self, x):
        """Returns backbone features (B, 128, 8) for dyn stream."""
        x = add_positional_encoding(x, self.pos_encoding)
        x = F.relu(self.conv1_dyn(x))
        x = F.relu(self.conv2_dyn(x))
        x = F.relu(self.conv3_dyn(x))
        x = F.relu(self.conv4_dyn(x))
        return x

    def forward_spatial(self, x):
        """
        Returns spatial feature map of shape (B, d_max, 128) from coord backbone.
        """
        features = self._forward_coord_backbone(x)
        x = self.conv_spatial(features)
        x = F.interpolate(x, size=128, mode='linear', align_corners=False)
        return x

    def forward(self, x):
        """
        Returns both streams:
        z_coord: soft centroids, shape (B, d_max)
        z_dyn: dynamics representations, shape (B, d_dyn)
        """
        # Coord stream
        coord_features = self._forward_coord_backbone(x)
        a_spatial = self.conv_spatial(coord_features)
        a_spatial = F.interpolate(a_spatial, size=128, mode='linear', align_corners=False)
        z_coord, _ = calculate_centroid_and_variance(a_spatial)

        # Dyn stream (independent backbone)
        dyn_features = self._forward_dyn_backbone(x)          # (B, 128, 8)
        a_dyn = self.conv_identity_dyn(dyn_features)           # (B, d_max, 8)
        z_dyn = a_dyn.mean(dim=-1)                             # (B, d_max)

        return z_coord, z_dyn

def calculate_centroid_and_variance(a_spatial):
    """
    Computes spatial centroid and soft spatial variance for each channel.
    Args:
        a_spatial (Tensor): shape (B, d_max, 128)
    Returns:
        x_mean (Tensor): shape (B, d_max)
        var (Tensor): shape (B, d_max)
    """
    p_c = F.softmax(a_spatial, dim=-1) # shape (B, d_max, 128)
    coords = torch.arange(128, device=a_spatial.device, dtype=torch.float32)
    x_mean = torch.sum(coords * p_c, dim=-1) # shape (B, d_max)
    var = torch.sum(((coords.unsqueeze(0).unsqueeze(1) - x_mean.unsqueeze(-1)) ** 2) * p_c, dim=-1) # shape (B, d_max)
    return x_mean, var

def add_positional_encoding(x, pos_encoding="none"):

    def __init__(self, d_max=8):
        """

        """
        super().__init__()

        self.conv1 = nn.Conv1d(3, 16, kernel_size=5, stride=2, padding=2)
        self.conv2 = nn.Conv1d(16, 32, kernel_size=5, stride=2, padding=2)
        self.conv3 = nn.Conv1d(32, 64, kernel_size=5, stride=2, padding=2)
        self.conv4 = nn.Conv1d(64, 128, kernel_size=5, stride=2, padding=2)
        
        # Dual stream heads
        self.conv_spatial_coord = nn.Conv1d(128, d_max, kernel_size=1)
        self.conv_spatial_dyn = nn.Conv1d(128, d_max, kernel_size=1)
        
    def forward_spatial(self, x):
        """
        Returns spatial feature map of shape (B, d_max, 128) for the coordinate stream.
        To ensure backprop through loss_spatial only flows to conv_spatial_coord,
        we detach the output of conv4 before passing to conv_spatial_coord.
        """
        x = F.relu(self.conv1(x))
        x = F.relu(self.conv2(x))
        x = F.relu(self.conv3(x))
        x = F.relu(self.conv4(x)) # (B, 128, 8)

        x_coord = self.conv_spatial_coord(x.detach())  # (B, d_max, 8)

        return x_coord

        x = F.relu(self.conv2(x))
        x = F.relu(self.conv3(x))
