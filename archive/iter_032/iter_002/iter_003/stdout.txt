class SeparateDynEncoder(nn.Module):
    """
    Encoder with TWO independent CNN backbones:
      (a) coord_backbone: conv1->conv2->conv3->conv4->conv_spatial
          Produces z_coord via soft-argmax centroid from spatial feature map.
      (b) dyn_backbone: conv1_dyn->conv2_dyn->conv3_dyn->conv4_dyn->conv_identity_dyn
          Same architecture, SEPARATE parameters.
          Produces z_dyn via mean pooling over the spatial dimension.

    Both backbones process the SAME RGB input independently.
    Each backbone adds positional encoding independently.
    """

    def __init__(self, d_max=8, pos_encoding="none", dyn_readout="mean",
                 sub_features=1, dyn_source="spatial"):
        super().__init__()
        self.d_max = d_max
        self.pos_encoding = pos_encoding
        # dyn_readout and sub_features accepted for interface compatibility,
        # but this encoder always uses mean readout on the dyn backbone.
        self.dyn_readout = dyn_readout
        self.sub_features = sub_features
        self.dyn_source = dyn_source

        if pos_encoding == "none":
            in_channels = 3
        elif pos_encoding == "linear":
            in_channels = 4]

        self.conv1_dyn = nn.Conv1d(in_channels, 16, kernel_size=5, stride=2, padding=2)

        self.conv3_dyn = nn.Conv1d(32, 64, kernel_size=5, stride=2, padding=2)

    @property
    def d_dyn(self):
        return self.d_max * self.sub_features

    def _forward_coord_backbone(self, x):
        """Returns backbone features (B, 128, 8) for coord stream."""
        x = add_positional_encoding(x, self.pos_encoding)
        x = F.relu(self.conv1(x))
        x = F.relu(self.conv2(x))
        x = F.relu(self.conv3(x))
        x = F.relu(self.conv4(x))
        return x

    def _forward_dyn_backbone(self, x):
        """Returns backbone features (B, 128, 8) for dyn stream."""
        x = add_positional_encoding(x, self.pos_encoding)
        x = F.relu(self.conv1_dyn(x))
        x = F.relu(self.conv2_dyn(x))
        x = F.relu(self.conv3_dyn(x))
        x = F.relu(self.conv4_dyn(x))
        return x

    def forward_spatial(self, x):
        """
        Returns spatial feature map of shape (B, d_max, 128) from coord backbone.
        """
        features = self._forward_coord_backbone(x)
        x = self.conv_spatial(features)
        x = F.interpolate(x, size=128, mode='linear', align_corners=False)
        return x

    def forward(self, x):
        """
        Returns both streams:
        z_coord: soft centroids, shape (B, d_max)
        z_dyn: dynamics representations, shape (B, d_dyn)
        """
        # Coord stream
        coord_features = self._forward_coord_backbone(x)
        a_spatial = self.conv_spatial(coord_features)
        a_spatial = F.interpolate(a_spatial, size=128, mode='linear', align_corners=False)
        z_coord, _ = calculate_centroid_and_variance(a_spatial)

        # Dyn stream (independent backbone)
        dyn_features = self._forward_dyn_backbone(x)          # (B, 128, 8)
        a_dyn = self.conv_identity_dyn(dyn_features)           # (B, d_max, 8)
        z_dyn = a_dyn.mean(dim=-1)                             # (B, d_max)

        return z_coord, z_dyn
