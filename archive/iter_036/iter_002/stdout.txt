def detect_ptr_obj_collisions(step, pre_ptr_vel, pre_obj_vels, post_ptr_vel, post_obj_vels,
                               ptr_pos, obj_positions, ptr_radius, obj_radii):
    """Detect pointer-object collisions by proximity + velocity change."""
    colls = []
    for i in range(N_OBJ):
        d = abs(ptr_pos - obj_positions[i])
        if d < ptr_radius + obj_radii[i] + PROX_THRESH:
            dv_ptr = post_ptr_vel - pre_ptr_vel
            dv_obj = post_obj_vels[i] - pre_obj_vels[i]
            if abs(dv_ptr) > DV_THRESH and abs(dv_obj) > DV_THRESH:
                colls.append((step, i, float(pre_ptr_vel), float(pre_obj_vels[i]),
                              float(post_ptr_vel), float(post_obj_vels[i]),
                              float(dv_ptr), float(dv_obj)))
    return colls


# ── POMLRE Metric ──
def compute_pomlre(collisions, true_masses):
    """Per-Object Median Log-Ratio Error."""
    errors = []
    per_obj_n_valid = []
    per_obj_errors = []
    
    for i in range(N_OBJ):
        obj_colls = [c for c in collisions if c[1] == i and abs(c[6]) > 1.0]  # |dv_obj| > 1.0
        # Note: c[6] = dv_obj, c[5] = dv_ptr
        # Actually c indices: (step, obj_i, pre_ptr, pre_obj, post_ptr, post_obj, dv_ptr, dv_obj)
        valid = [c for c in obj_colls if abs(c[7]) > 1.0]  # |Δv_obj| > 1.0
        
        if len(valid) >= 3:
            m_ests = [-PTR_MASS * c[6] / c[7] for c in valid]  # -M_ptr * dv_ptr / dv_obj
            m_hat = float(np.median(m_ests))
            if m_hat > 0 and true_masses[i] > 0:
                err = abs(np.log(m_hat / true_masses[i]))
            else:
                err = 2.0
        elif len(valid) >= 1:
            m_ests = [-PTR_MASS * c[6] / c[7] for c in valid]
            m_hat = float(np.mean(m_ests))
            if m_hat > 0 and true_masses[i] > 0:
                err = abs(np.log(m_hat / true_masses[i]))
            else:
                err = 2.0
        else:
            err = 2.0
        
        errors.append(err)
        per_obj_n_valid.append(len(valid))
        per_obj_errors.append(err)
    
    return float(np.mean(errors)), per_obj_n_valid, per_obj_errors


# ── Bootstrap CI ──
def bootstrap_ci(a_vals, b_vals, n_boot=N_BOOT):
    """Paired bootstrap CI for mean(b) - mean(a). Returns (lower_95, upper_95)."""
    n = len(a_vals)
    gaps = []
    for _ in range(n_boot):
        idx = np.random.randint(0, n, size=n)
        gaps.append(float(np.mean(b_vals[idx]) - np.mean(a_vals[idx])))
    gaps = np.array(gaps)
    return float(np.percentile(gaps, 2.5)), float(np.percentile(gaps, 97.5))

    obj_pushes = [0, 0, 0]  # per-object push count
    
    for step in range(N_STEPS):
        # Save pre-state
        pre_ptr_vel = env.pointer_vel
        pre_obj_vels = env.velocities.copy()
        
        # Compute action
        if condition == 'ORACLE':
            target = int(np.argmin(obj_ptr_colls))
            error = env.positions[target] - env.pointer_pos
            d_error = (error - prev_err) if prev_err is not None else 0.0
            prev_err = error
            acc = KP * error + KD * d_error
            
            if push_cd > 0:
                push_cd -= 1
            
            do_push = False
            if abs(error) <= PUSH_DIST and push_cd == 0 and pushes_left > 0:

                pushes_left -= 1
                pushes_used += 1
                obj_pushes[target] += 1

                # Switch target to next least-collided

                counts[target] = np.inf

                prev_err = None

                env.pointer_vel = PUSH_VEL * np.sign(error)

            do_push = rng.rand() < 0.1 and pushes_left > 0
            if do_push:
                pushes_left -= 1
                pushes_used += 1
            action = {'acc': acc, 'push': do_push}
            
        else:  # PASSIVE
            action = {'acc': 0.0, 'push': False}
        
        obs, info = env.step(action)
        
        post_ptr_vel = env.pointer_vel
        post_obj_vels = info['velocities'].copy()
        
        if not (0 <= env.pointer_pos <= 128):
            ptr_oob += 1
        
        # Detect collisions
        step_colls = detect_ptr_obj_collisions(
            step, pre_ptr_vel, pre_obj_vels, post_ptr_vel, post_obj_vels,
            env.pointer_pos, env.positions, env.pointer_radius, env.radii

        all_colls.extend(step_colls)
        
        # Update ORACLE collision counts

temp_rad[i]\n                    if temp_vel[i] > 0.0:\n                        temp_vel[i] = -temp_vel[i]\n\n            # Unpack\n            self.positions = temp_pos[:-1].copy()\n            self.velocities = temp_vel[:-1].copy()\n            self.pointer_pos = temp_pos[-1]\n            self.pointer_vel = temp_vel[-1]\n\n        # Increment step counter (for probe event recording)\n        self._step_counter = getattr(self, \'_step_counter\', 0) + 1\n\n        # Noisy-TV / Structured Distractor (inherited logic)\n        if self.noisy_tv:\n            step_noise = np.random.normal(0.0, 2.0)\n            self.noisy_tv_pos += step_noise\n            self.noisy_tv_pos = np.clip(\n                self.noisy_tv_pos, self.noisy_tv_radius, 128.0 -     """

            d_error = (error - prev_err) if prev_err is not None else 0.0

### Structure

1. **`FoveatedGazeSandbox`** — Subclass of `PhysicsSandbox`:
   - **Ghostly pointer**: The substep loop skips any collision pair where `i == PTR_IDX` or `j == PTR_IDX`, so the pointer never collides with objects during physics integration. Pointer still bounces off walls at 0/128 and receives `acc` acceleration.
   - **Probe mechanism**: Applied *before* the substep loop. Finds the nearest object within `gaze_radius=8` of the gaze center. If found, applies a 1D elastic collision (gaze mass=10.0 × object mass), updates both velocities, decrements `probe_budget` (starts at 20). Records each event as `(step, obj_idx, v_gaze_pre, v_obj_pre, v_gaze_post, v_obj_post)`. Budget decrements even on misses.
   - **Delta-v definition**: `pre` velocities recorded before probe + substep loop; `post` velocities after the full substep loop.
   - **Render**: Applies a foveation mask zeroing pixels outside `[gaze_pos ± gaze_radius]`.

2. **`run_episode(condition, seed, arm)`**:
   - `arm='A'` (normal obj-obj) or `'B'` (pass-through obj-obj)
   - All conditions run 2000 steps with `probe_budget=20`
   - **ORACLE**: PD-tracks least-probed object (Kp=2.0, Kd=0.5). Fires probe when target within gaze radius AND `|error| ≤ 6.0`. Switches target after each successful probe.
   - **RANDOM**: `acc ∈ [-10,10]`, probe with p=0.01 per step until budget exhausted.
   - **PASSIVE**: Zero acceleration, zero probing → POMLRE = 2.0.
   - Uses deterministic RNG: `np.random.RandomState(seed * 1000 + hash(condition + arm) % 10000)`.

3. **POMLRE + branch tracking** (`compute_pomlre_and_branches`):
   - Filters probe events with `|Δv_obj| > 1.0`, computes `m_est = -M_gaze × Δv_gaze / Δv_obj`
   - Branch 0: 0 valid events → error = 2.0
   - Branch 1: 1-2 valid events → mean of estimates → `|log(m_hat / m_true)|`
   - Branch 2: ≥3 valid events → median of estimates → `|log(m_hat / m_true)|`
   - Tracks `(seed, obj_idx) → branch` for distribution analysis.

4. **Main flow**:
   - CV Gate on 5 seeds × 2 arms (RANDOM condition); arms failing CV ≥ 0.5 are excluded
   - Full bracket on 8 seeds × surviving arms × 3 conditions
   - Gates F1–F4 + estimation-only decomposition
   - Outputs: `per_run.csv`, `cv_gate.txt`, `sanity_checks.txt`, `analysis.md`, `branch_distribution.txt`
